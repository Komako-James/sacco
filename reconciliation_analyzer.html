SELECT account_code, account_name, account_type, account_category
FROM chart_of_accounts
WHERE account_code IN ('2010','2020','1010','1020');
SELECT entry_id AS id, entry_date, ledger_code, ledger_name, debit, credit, description, transaction_reference, receipt_number, status
FROM ledger_entries
WHERE transaction_reference IN ('RP20260617185109210','RP20260617185239943')
ORDER BY entry_date, entry_id;
SELECT entry_id AS id, entry_date, ledger_code, ledger_name, debit, credit, description, transaction_reference, receipt_number, status, member_id
FROM ledger_entries
WHERE transaction_reference IN ('STR-3-4-1780077242','STR-3-4-1780077070')
ORDER BY entry_date, entry_id;
-- Option A: single number
